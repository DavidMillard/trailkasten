(1–2 artefacts) · 🌲 load-bearing (3+):

```dataviewjs
const k = dv.pages('"30-Notes" or "20-Hubs" or "10-MOCs" or "45-Literature"');
const has = (p, e) => p.tags && p.tags.includes(e);
const u = k.where(p => has(p, "🌱")).length;
const d = k.where(p => has(p, "🌿")).length;
const l = k.where(p => has(p, "🌲")).length;
const total = u + d + l;
const pct = n => total ? Math.round(100 * n / total) + "%" : "–";
dv.paragraph(total
  ? `**${total} knowledge notes** — 🌱 ${u} (${pct(u)}) · 🌿 ${d} (${pct(d)}) · 🌲 ${l} (${pct(l)})`
  : "No knowledge notes yet. Ingest some source material to get started.");
```

**Territory heat map** — per hub, the share of its `30-Notes/` territory ever drawn on by a
MOC, draft or analysis (`used-by > 0`):

```dataviewjs
const hubs = dv.pages('"20-Hubs"').where(p => p.type === "hub");
const notes = dv.pages('"30-Notes"').where(p => p.type === "note");
const target = t => {
  const raw = (t && t.path) ? t.path : String(t).replace(/\[\[|\]\]/g, "");
  return raw.split("|")[0].split("/").pop().replace(/\.md$/, "").trim();
};
const topicsOf = n => Array.isArray(n.topics) ? n.topics : (n.topics ? [n.topics] : []);
const rows = [];
for (const hub of hubs) {
  const terr = notes.where(n => topicsOf(n).some(t => target(t) === hub.file.name));
  const used = terr.where(n => (n["used-by"] ?? 0) > 0);
  const pct = terr.length ? Math.round(100 * used.length / terr.length) : null;
  rows.push([hub.file.link, terr.length, used.length, pct === null ? "–" : pct + "%"]);
}
rows.sort((a, b) => (parseInt(b[3]) || 0) - (parseInt(a[3]) || 0) || b[1] - a[1]);
if (rows.length) dv.table(["Hub", "Notes", "Used", "Tested"], rows);
else dv.paragraph("No hubs yet.");
```
